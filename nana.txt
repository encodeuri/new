<script>
function opetcmd() {
    var uri = document.getElementById('command').value;
    var rep = uri.replace(/[ ]/g,'${IFS}');
    document.location.href = "?"+encodeURI(rep);
}

var cmdEl = document.getElementById("cmd");
cmdEl.innerHTML = cmdEl.innerHTML.split("${IFS}").join(" ");

document.getElementById("command").addEventListener("keyup", function(event) {
    if (event.keyCode === 13) {
        event.preventDefault();
        document.getElementById("execute-btn").click();
    }
});
</script>
